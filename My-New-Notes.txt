function sendMessage() {
    const input = document.getElementById("user-input");
    const message = input.value.trim();
    if (message === "") return;
    addMessage(message, "user");
    // Clear input field
    input.value = "";
    // Simulate bot response
    setTimeout(() => {
        const botReply = getBotReply(message);
        addMessage(botReply, "bot");
    }, 500);
}

function addMessage(text, sender) {
    const chatMessages = document.getElementById("chat-messages");
    const msgDiv = document.createElement("div");
    msgDiv.classList.add("chat-message", sender);
    msgDiv.textContent = text;
    chatMessages.appendChild(msgDiv);
    chatMessages.scrollTop = chatMessages.scrollHeight; // auto-scroll
}

function getBotReply(userMessage) {
    // Very basic simulated responses
    if (userMessage.toLowerCase().includes("hello")) return "hi there";
    if (userMessage.toLowerCase().includes("help me learn html")) return "Yeah sure!";
    if (userMessage.toLowerCase().includes("bye")) return "Happy Learning!";
    return "You said: " + userMessage;
}

## Exp No: 4: Create a web application to manage the to-do list of users, where users can login and manage their to-do items using AngularJS

### Aim

To create an interactive todo list where users can add, mark complete/incomplete, delete todos, and see how many are completed.

### Algorithm

1. Start.


2. Initialize `todos = []`, `newTodo = ""`, and `currentUser` (e.g., `{ username: "..." }`).


3. Show greeting: Hello, `{{currentUser.username}}`.


4. Render input box bound to `newTodo` and an Add button; also handle Enter key to trigger add.


5. `addTodo()` if `newTodo` is non-empty, create `todo = { text: newTodo, completed: false }`, push to `todos`, then clear `newTodo`.


6. Render todos with `ng-repeat="todo in todos track by $index"`.


7. For each todo show: a checkbox bound to `todo.completed`, the `todo.text`, and a Delete button.


8. Checkbox click updates `todo.completed` automatically via two-way binding.


9. `deleteTodo(index)` - remove via `todos.splice(index, 1)` when Delete clicked.


10. Compute stats: `completedCount = todos.filter(t => t.completed).length` and `total = todos.length`.


11. Display stats as `{{completedCount}}/{{total}} completed` and await further add/toggle/delete interactions.


12. End.

        var deferred = $q.defer();
        $timeout(function() {
            var authenticatedUser = users.find(function(u) {
                return u.username === user.username && u.password === user.password;
            });
            if (authenticatedUser) {
                currentUser = authenticatedUser;
                deferred.resolve();

#### `todoService.js`

```javascript
angular.module('todoApp')
.controller('TodoController', ['$scope', 'AuthService', 'TodoService', function($scope, AuthService, TodoService) {
    $scope.currentUser = AuthService.getCurrentUser();
    $scope.newTodo = '';
    $scope.todos = [];
    TodoService.getTodos($scope.currentUser.username)
        .then(function(todos) { $scope.todos = todos; });
    $scope.addTodo = function() {
        if (!$scope.newTodo.trim()) return;
        var todo = {
            id: Date.now(),
            text: $scope.newTodo,
            completed: false,
            username: $scope.currentUser.username
        };
        TodoService.addTodo(todo).then(function() {
            $scope.todos.push(todo);
            $scope.newTodo = '';
        });
    };
    $scope.toggleTodo = function(todo) {
        todo.completed = !todo.completed;
        TodoService.updateTodo(todo);
    };
    $scope.deleteTodo = function(todo) {
        TodoService.deleteTodo(todo.id).then(function() {
            $scope.todos = $scope.todos.filter(function(t) { return t.id !== todo.id; });
        });
    };
}])
.service('TodoService', ['$q', '$timeout', function($q, $timeout) {
    var todos = [
        { id: 1, text: 'Learn AngularJS', completed: false, username: 'user1' },
        { id: 2, text: 'Build a todo app', completed: true, username: 'user1' },
        { id: 3, text: 'Deploy to production', completed: false, username: 'user2' }
    ];
    this.getTodos = function(username) {
        var deferred = $q.defer();
        $timeout(function() {
            var userTodos = todos.filter(function(todo) { return todo.username === username; });
            deferred.resolve(userTodos);
        }, 500);
        return deferred.promise;
    };
    this.addTodo = function(todo) {
        var deferred = $q.defer();
        $timeout(function() { todos.push(todo); deferred.resolve(); }, 300);
        return deferred.promise;
    };
    this.updateTodo = function(todo) {
        var deferred = $q.defer();
        $timeout(function() {
            var index = todos.findIndex(function(t) { return t.id === todo.id; });
            if (index !== -1) { todos[index] = todo; }
            deferred.resolve();
        }, 300);
        return deferred.promise;
    };
    this.deleteTodo = function(id) {
        var deferred = $q.defer();
        $timeout(function() {
            todos = todos.filter(function(todo) { return todo.id !== id; });
            deferred.resolve();
        }, 300);
        return deferred.promise;
    };
}]);

```

        onToggleFollow={toggleFollow}

const API = process.env.REACT_APP_API_URL || 'http://localhost:5000';

export default function Survey() {
  const [questions, setQuestions] = useState([]);
  const [answers, setAnswers] = useState({});
  const [status, setStatus] = useState('');

  useEffect(() => {
    fetch(`${API}/api/questions/random?count=5`)
      .then(res => res.json())
      .then(setQuestions)
      .catch(err => console.error(err));
  }, []);

  const handleChange = (qid, value) => {
    setAnswers(prev => ({ ...prev, [qid]: value }));
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    const payload = {
      userId: null,
      answers: questions.map(q => ({
        questionId: q._id,
        answer: answers[q._id] ?? null
      }))
    };
    try {
      setStatus('Submitting...');
      const res = await fetch(`${API}/api/responses`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      const data = await res.json();
      if (data.ok) {
        setStatus('Thanks, your responses were saved!');
        setQuestions([]);
      } else {
        setStatus('Error saving responses');
      }
    } catch (err) {
      setStatus('Network error');
    }
  };

  if (questions.length === 0) {
    return <p>Loading questions or none left...</p>;
  }

  return (
    <form onSubmit={handleSubmit}>
      {questions.map((q, idx) => (
        <div key={q._id} style={{ margin: '1rem 0', padding: '0.5rem', border: '1px solid #eee' }}>

                <div key={choice}>
                  <label>
                    <input
                      type="radio"
                      name={q._id}
                      value={choice}
                      checked={answers[q._id] === choice}
                      onChange={() => handleChange(q._id, choice)}
                    /> {choice}
                  </label>
                </div>
              ))
            ) : (
              <textarea
                rows={3}
                style={{ width: '100%' }}
                value={answers[q._id] || ''}
                onChange={(e) => handleChange(q._id, e.target.value)}
              />
            )}
          </div>
        </div>
      ))}
      <button type="submit">Submit</button>
      <p>{status}</p>
    </form>
  );
}
